#include<iostream>
using namespace std;

// FCFS Scheduling function
void FCFS_Scheduling(int n, int AT[], int BT[], string process[], int CT[],int TAT[], int WT[])
{
    // Step 1: Sort processes according to Arrival Time (AT)
    for(int i = 0; i < n-1; i++)
    {
        for(int j = i+1; j < n; j++)
        {
            if(AT[i] > AT[j])
            {
                // Swap Arrival Time
                int temp1 = AT[i];
                AT[i] = AT[j];
                AT[j] = temp1;
                // Swap Burst Time accordingly
                int temp2 = BT[i];
                BT[i] = BT[j];
                BT[j] = temp2;
                // Swap process names to maintain mapping
                string temp3 = process[i];
                process[i] = process[j];
                process[j] = temp3;
            }
        }
    }

    // Step 2: Calculate Completion Time (CT)
    for(int i=0; i<n; i++)
    {
        if(i==0)
        {
            // First process completes at AT + BT
            CT[i] = AT[i] + BT[i];
        }
        else
        {
            // Next process completes after previous one finishes
            CT[i] = CT[i-1] + BT[i];
        }
    }

    // Step 3: Calculate Turnaround Time (TAT)
    for(int i=0; i<n; i++)
    {
        // TAT = Completion Time - Arrival Time
        TAT[i] = CT[i]-AT[i];
    }

    // Step 4: Calculate Waiting Time (WT)
    for(int i=0; i<n; i++)
    {
        // WT = Turnaround Time - Burst Time
        WT[i] = TAT[i]-BT[i];
    }
}

int main() {
    int n;
    cout << "Enter no. of process : ";
    cin>>n;

    // Arrays for Burst Time, Arrival Time, Completion Time, etc.
    int BT[n], AT[n], CT[n],TAT[n],WT[n];
    string process[n];

    cout<<"Enter processes : ";
    for(int i=0; i<n; i++)
    {
        CT[i] = 0;
        TAT[i] = 0;
        WT[i] = 0;
        cin>>process[i];
    }

    cout<<"Enter Burst time : ";
    for(int i=0; i<n; i++)
        cin>>BT[i];

    cout<<"Enter Arrival time : ";
    for(int i=0; i<n; i++)
        cin>>AT[i];

    FCFS_Scheduling(n, AT, BT, process, CT, TAT, WT);

    cout << "\n\nProcess\tBT\tAT\tCT\tTAT\tWT\n";
    cout << "-----------------------------------------\n";

    int sum1 = 0, sum2 = 0;

    {
        cout << process[i] << "\t"
             << BT[i] << "\t"
             << AT[i] << "\t"
             << CT[i] << "\t"
             << TAT[i] << "\t"
             << WT[i] << endl;
        sum1 += TAT[i];
        sum2 += WT[i];
    }

    cout << "\nAverage Turn Around Time : " << (double)sum1/n << endl;
    cout << "Average Waiting Time : " << (double)sum2/n << endl;

    return 0;
}

// SJF Scheduling function
void sjf_Scheduling(int n, int AT[], int BT[], string process[], int CT[],int TAT[], int WT[])
{
    // handling for idle cpu case
    int c=0;
    bool flag =0;
    for(int i=0; i<n; i++)
        if(AT[i]!=0) {
            flag = 1;
            break;
        }
    if(flag==1)
        CT[0] = AT[0] + BT[0];
    c=CT[0];

    // Sort on the basis of burst time
    for(int i = 0; i < n-1; i++)
    {
        for(int j = i+1; j<n; j++)
        {
            if(BT[i] > BT[j])
            {
                int temp1 = BT[i]; BT[i] = BT[j]; BT[j] = temp1;
                int temp2 = AT[i]; AT[i] = AT[j]; AT[j] = temp2;
                string temp3 = process[i]; process[i] = process[j]; process[j] = temp3;
                int temp4 = CT[i]; CT[i] = CT[j]; CT[j] = temp4;
            }
        }
    }

    // Calculate Completion Time (CT)
    for(int i=0; i<n; i++)
    {
        if(AT[i] == 0)
        {
            CT[i] = AT[i] + BT[i];
            c=CT[i];
            break;
        }
    }

    for(int i=0; i<n; i++)
    {
        if(CT[i] == 0)
        {
            CT[i] = c + BT[i];
            c = CT[i];
        }
    }

    // Calculate TAT and WT
    for(int i=0; i<n; i++)
        TAT[i] = CT[i]-AT[i];
    for(int i=0; i<n; i++)
        WT[i] = TAT[i]-BT[i];
}

// priority based Scheduling function
void pb_scheduling(int n, int AT[], int BT[], string process[], int CT[],int TAT[], int WT[],string p,int priority[])
{
    // handling for idle cpu case
    int c=0;
    bool flag =0;
    for(int i=0; i<n; i++)
        if(AT[i]!=0) { flag = 1; break; }
    if(flag==1) {
        CT[0] = AT[0] + BT[0];
        c=CT[0];
    }

    // Sort on the basis of priority
    for(int i = 0; i < n-1; i++)
    {
        for(int j = i+1; j<n; j++)
        {
            // 'B'/'b' = bigger number is higher priority
            // 'S'/'s' = smaller number is higher priority
            if( ((int(p[0]) == 66 || int(p[0]) == 98) && (priority[i] < priority[j])) ||
                    ((int(p[0]) == 83 || int(p[0]) == 115) && (priority[i] > priority[j])) )
            {
                int temp = priority[i]; priority[i] = priority[j]; priority[j] = temp;
                int temp1 = BT[i]; BT[i] = BT[j]; BT[j] = temp1;
                int temp2 = AT[i]; AT[i] = AT[j]; AT[j] = temp2;
                string temp3 = process[i]; process[i] = process[j]; process[j] = temp3;
                int temp4 = CT[i]; CT[i] = CT[j]; CT[j] = temp4;
            }
        }
    }

    // Calculate Completion Time (CT)
    for(int i=0; i<n; i++)
    {
        if(AT[i] == 0)
        {
            CT[i] = AT[i] + BT[i];
            c=CT[i];
            break;
        }
    }
    for(int i=0; i<n; i++)
    {
        if(CT[i] == 0) {
            CT[i] = c + BT[i];
            c = CT[i];
        }
    }

    // Calculate TAT and WT
    for(int i=0; i<n; i++)
        TAT[i] = CT[i]-AT[i];
    for(int i=0; i<n; i++)
        WT[i] = TAT[i]-BT[i];
}

#include<iostream>

// FIRST FIT FUNCTION
void first_fit(int n1, int n2, int Mem_block[], int process[], int block[])
{
    for(int i=0; i<n2; i++)
    {
        for(int j=0; j<n1; j++)
        {
            if(Mem_block[j]>=process[i])
            {
                Mem_block[j] = Mem_block[j] - process[i];
                block[i] = j+1;
                break;
            }
        }
    }
}

// BEST FIT FUNCTION
void best_fit(int n1, int n2, int Mem_block[], int process[], int block[],int b[])
{
    // Sort memory blocks in ASCENDING order
    for(int i=0; i<n1-1; i++)
    {
        for(int j=i+1; j<n1; j++)
        {
            if(Mem_block[j] <= Mem_block[i])
            {
                int temp = Mem_block[i]; Mem_block[i] = Mem_block[j]; Mem_block[j] = temp;
                int temp1 = b[i]; b[i] = b[j]; b[j] = temp1;
            }
        }
    }
    for(int i=0; i<n2; i++)
    {
        for(int j=0; j<n1; j++)
        {
            if(Mem_block[j]>=process[i])
            {
                Mem_block[j] = Mem_block[j] - process[i];
                block[i] = b[j];
                break;
            }
        }
    }
}

void worst_fit(int n1, int n2, int Mem_block[], int process[], int block[],int b[])

#include<iostream>
using namespace std;

// FIRST FIT FUNCTION
void first_fit(int n1, int n2, int Mem_block[], int process[], int block[])
{
    // Loop through each process
    for(int i=0; i<n2; i++)
    {
        // Check each memory block from the beginning
        for(int j=0; j<n1; j++)
        {
            // If block size is enough for process
            if(Mem_block[j]>=process[i])
            {
                // Allocate: reduce block size by process size
                Mem_block[j] = Mem_block[j] - process[i];
                // Store block number (1-indexed)
                block[i] = j+1;
                // Move to next process
                break;
            }
        }
    }
}

int main() {
    int n1, n2;
    cout << "Enter no. of memory blocks : "; cin >> n1;
    cout << "Enter no. of processes : "; cin >> n2;

    int Mem_block[n1], process[n2], block[n2];

    cout << "Enter Memory Block sizes (KB) :\n";
    for(int i=0; i<n1; i++) {
        block[i] = 0;  // Initialize to 0 (not allocated)
        cin >> Mem_block[i];
    }

    cout << "Enter Process sizes (KB) :\n";
    for(int i=0; i<n2; i++)
        cin >> process[i];

    first_fit(n1, n2, Mem_block, process, block);

    cout << "\nFIRST FIT RESULT\n";
    cout << "Process\tSize\tBlock Allocated\n";
    cout << "----------------------------\n";

// BEST FIT FUNCTION
void best_fit(int n1, int n2, int Mem_block[], int process[], int block[], int b[])
{
    // Sort memory blocks in ASCENDING order (smallest first = best fit)
    for(int i=0; i<n1-1; i++)
    {
        for(int j=i+1; j<n1; j++)
        {
            if(Mem_block[j] <= Mem_block[i])
            {
                // Swap block sizes
                int temp = Mem_block[i];
                Mem_block[i] = Mem_block[j];
                Mem_block[j] = temp;
                // Swap original block indices
                int temp1 = b[i];
                b[i] = b[j];
                b[j] = temp1;
            }
        }
    }

    // Allocate processes — smallest fitting block first
    for(int i=0; i<n2; i++)
    {
        for(int j=0; j<n1; j++)
        {
            if(Mem_block[j]>=process[i])
            {
                Mem_block[j] = Mem_block[j] - process[i];
                block[i] = b[j];  // Store original block number
                break;
            }
        }
    }
}

             << pid << " PID = " << getpid() << endl;

📄 fork_wait.cpp  —  iii) Before terminating the parent wait for the child to finish its task
#include <iostream>
#include <unistd.h>
#include <sys/wait.h>
using namespace std;
int main(){
    cout << "starting program. Parent PID: " << getpid() << "\n";
    pid_t pid = fork();
    if(pid == 0){
        cout << "\n[child] I am child process. My PID is "
             << getpid() << endl;
        cout << "[child] I am starting a task takes 2 seconds..."
             << endl;
    }
    else if(pid > 0){
        cout << "[Parent] I created a child with PID : "
             << getpid() << endl;
        cout << "[Parent] I will now wait for the child to finish its task."
             << endl;
        wait(NULL);
        cout << "\n[Parent] child process is finished. I am now terminating as well!"
             << endl;
    }
    else{
        cout << "fork failed" << endl;
        return 1;
    }
    return 0;
}
